def noise_from(n, extra=None, exclude=None):
    extra = extra or []
    exclude = set(exclude or [])
    prefixes = (
        "docs/gallery", "docs/assets/", "docs/gallery.html",
        "archify.zip", "archify/examples/", "examples/",
        "generated/", "archify/package-lock.json",
        "archify/renderers/shared/generated-",
        "docs/cases/",
    )
    exact = {
        "docs/gallery.html", "archify.zip",
        "archify/package-lock.json",
        "docs/assets/archify-live-proof.gif",
        "docs/assets/archify-live-proof.json",
    }
    out = []
    for p in receipts[n]:
        if p in exclude:
            continue
        if p in extra or p in exact or p.startswith(prefixes) or any(p.startswith(pref) for pref in prefixes):
            out.append(p)
            continue
        if p.endswith((".html",".png",".gif",".webp",".webm",".jpg")) and (
            p.startswith("docs/") or p.startswith("examples/") or p.startswith("experiments/")
        ):
            out.append(p)
    # extras not already in list
    for p in extra:
        if p not in out and p not in exclude:
            out.append(p)
    return sorted(set(out))
